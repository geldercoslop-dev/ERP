### 3. client/src/lib/security/sessionToken.ts ✓
**Status:** DESIGN CORRETO (memory-only tokens)

```typescript
/** Token de sessão em memória volátil (não persiste em localStorage) */

import { getCSRFToken, getCSRFHeaderName } from "./csrfToken";

export const SESSION_TOKEN_STORAGE_KEY = "grs-session-token";

// Token em memória apenas (volátil)
let memoryToken: string | null = null;

export function setSessionToken(token: string | null): void {
  if (typeof window === "undefined") return;
  // Apenas memória volátil - não persiste em localStorage
  memoryToken = token;
}

export function getSessionToken(): string | null {
  if (typeof window === "undefined") return null;
  // Retorna da memória volátil
  return memoryToken;
}

/** Headers enviados em todas as chamadas à API da mesma origem. */
export function buildSessionHeaders(): Record<string, string> {
  const token = getSessionToken();
  const csrfToken = getCSRFToken();

  const headers: Record<string, string> = {};

  // Session token headers
  if (token) {
    headers["x-session-token"] = token;
    headers["X-Session-Token"] = token;
    headers["Authorization"] = `Bearer ${token}`;
  }

  // CSRF token header (required)
  if (csrfToken) {
    headers[getCSRFHeaderName()] = csrfToken;
  }

  // SECURITY: x-app-secret NUNCA deve ser enviado pelo frontend.
  // A autenticação do browser usa apenas JWT (Authorization: Bearer) + CSRF.
  // APP_SECRET é exclusivo do servidor e não pode ser exposto no bundle JS.

  return headers;
}
```

✓✓✓ PERFEITO
✓ Token em memória (volátil)
✓ Não persiste em localStorage
✓ Enviado via header (não exposto em HTML)
✓ Comentário de segurança claro
