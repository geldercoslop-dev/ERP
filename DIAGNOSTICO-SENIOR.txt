✅ PASSO 1: INICIAR MYSQL
━━━━━━━━━━━━━━━━━━━━━━━

Opção A: MySQL instalado localmente
────────────────────────────────
# Windows - Iniciar serviço
net start MySQL80

# Ou via Services
services.msc → encontrar MySQL → Start

# macOS
brew services start mysql

# Linux
sudo systemctl start mysql


Opção B: MySQL via Docker
────────────────────────────
docker run -d \
  --name mysql \
  -e MYSQL_ROOT_PASSWORD=root \
  -e MYSQL_DATABASE=vendas_app \
  -e MYSQL_USER=vendas \
  -e MYSQL_PASSWORD=Vendas_Prod_Secure2026 \
  -p 3306:3306 \
  mysql:latest


Opção C: MySQL via Docker Compose
────────────────────────────────
# Criar docker-compose.yml:
version: '3'
services:
  mysql:
    image: mysql:latest
    environment:
      MYSQL_ROOT_PASSWORD: root
      MYSQL_DATABASE: vendas_app
      MYSQL_USER: vendas
      MYSQL_PASSWORD: Vendas_Prod_Secure2026
    ports:
      - "3306:3306"
    volumes:
      - db_data:/var/lib/mysql

  redis:
    image: redis:latest
    ports:
      - "6379:6379"

volumes:
  db_data:

# Depois rodar:
docker-compose up -d
