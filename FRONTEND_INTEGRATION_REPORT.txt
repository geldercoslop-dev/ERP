PASSO 2: Usar em Componentes
──────────────────────────────

import { useEffect } from 'react';
import { listarClientes } from '@/services/clientService';
import { useRequest, Loading, ErrorDisplay } from '@/hooks/useRequest';
import { useAuthContext } from '@/contexts/AuthContext';

export default function Clientes() {
  const { isAuthenticated } = useAuthContext();
  const { data, isLoading, error, execute } = useRequest(null);

  useEffect(() => {
    if (!isAuthenticated) return;
    execute(() => listarClientes());
  }, [isAuthenticated]);

  if (isLoading) return <Loading />;
  if (error) return <ErrorDisplay error={error} />;

  return (
    <div>{data?.items.map(c => <div key={c.id}>{c.nome}</div>)}</div>
  );
}
